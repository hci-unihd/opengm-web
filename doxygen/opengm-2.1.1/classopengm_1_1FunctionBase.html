call a functor for each value of the function (in lexicographical order of the variable indices)

Example: 

    template<class T>
    struct MaxFunctor {
       void operator()(const T v) {
          if(v > max_)
             max_ = v;
       }
       T max_;
    };
    MaxFunctor<float> maxFunctor;
    maxFunctor.max_ = 0.0;
    function.forAllValuesInOrder(maxFunctor);
